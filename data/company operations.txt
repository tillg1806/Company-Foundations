# Company Foundations

mission "Company Foundations: Manager Optimize local Shuttle Route"
	name "Manager Fleet Optimization"
	invisible
	landing
	repeat
	to offer
		has "cf: shuttle"
		has "cf: managed"
		not "cf: hq suspended"
		has "cf: shuttle"
		not "cf: manager local optimized"
		has "cf: ship available: Blackbird"
		has "cf: ship available: Heavy Shuttle"
		"cf: reserve" >= 6340000
	on offer
		"cf: reserve" -= 3170000
		set "cf: manager local optimized"
		"cf: shuttle fleet" -= "cf: shuttle local ships"
		"cf: shuttle local pax" = 48
		"cf: shuttle local ships" = 3
		"cf: shuttle local daily crew" = 500
		"cf: shuttle local trip payout" = 11200
		"cf: shuttle local trip expenses" = 666
		"cf: shuttle fleet" += 3
		"cf: fleet value" += 3170000
		set "cf: shuttle local luxury"
		log "Company Foundations" "Manager Investment" `The operations manager optimized the local route with 1 Blackbird and 2 Heavy Shuttles for 3.170.000 company credits after keeping at least double the package cost in reserve.`

mission "Company Foundations: Allocate Positive Net Profit"
	name "Company Profit Allocation"
	invisible
	landing
	repeat
	to offer
		has "cf: active"
		"cf: unallocated net" > 0
	on offer
		"cf: allocation owner payout" = "cf: unallocated net" * "cf: payout share" / 100
		"cf: last owner payout" = "cf: allocation owner payout"
		"cf: last retained earnings" = "cf: unallocated net" - "cf: allocation owner payout"
		"cf: owner payable" += "cf: allocation owner payout"
		"cf: reserve" -= "cf: allocation owner payout"
		"cf: total owner allocations" += "cf: allocation owner payout"
		"cf: month owner allocations" += "cf: allocation owner payout"
		"cf: total retained earnings" -= "cf: allocation owner payout"
		"cf: month retained earnings" -= "cf: allocation owner payout"
		"cf: unallocated net" = 0
		clear "cf: distress open"
